SELECT o.id, o.name, r.name AS region, count(i.id) AS inspections, max(i.inspected_at) AS last_inspection FROM objects o JOIN regions r ON r.id = o.region_id LEFT JOIN inspections i ON i.object_id = o.id WHERE o.status = 'active' AND o.updated_at > now() - interval '30 days' GROUP BY o.id, r.name ORDER BY inspections DESC LIMIT 50;
